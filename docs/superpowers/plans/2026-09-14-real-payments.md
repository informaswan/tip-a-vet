> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn Tip a Vet from a static front-end mockup into a real product: vets onboard with a linked Venmo/PayPal/Cash App handle and get manually verified, tippers send the vet's tip directly through that service (Tip a Vet never touches it), and Tip a Vet earns revenue only through a separate, optional, Stripe-charged "support Tip a Vet" donation.

**Architecture:** The existing static site (plain HTML/CSS/JS, GitHub Pages, no build step) stays as-is and gains a Supabase backend (Postgres + Auth + Storage + two Edge Functions) reached directly from the browser via the Supabase JS client. Only two secrets — the Stripe secret key and the Supabase service-role key — ever leave the browser's reach, living solely in Edge Function environment variables.

**Tech Stack:** Vanilla JS (no framework/build step, matching the existing codebase), Supabase (Postgres, Auth, Storage, Edge Functions in Deno/TypeScript), Stripe Checkout (server-side session creation only — no Stripe.js needed client-side), Supabase CLI for local dev and deploys.

**Spec:** `docs/superpowers/specs/2026-09-14-real-payments-design.md`

## Global Constraints

- Tip a Vet must never store bank account/routing numbers or custody tip funds — the vet's tip always goes directly to their own Venmo/PayPal/Cash App.
- No build step, no framework — new frontend files are plain `<script>`-tag JS matching the existing style (`function` declarations, string concatenation, `const`/`let`, no JSX/TS in the browser).
- Only the Supabase URL and anon/public key are safe to commit client-side (protected by Row Level Security). The Stripe secret key, Supabase service-role key, and Stripe webhook signing secret must only ever be set as Supabase Edge Function secrets, never in any committed file.
- Every table gets Row Level Security enabled before any policy or data touches it — RLS is the only access-control boundary in this system (there is no app-level auth server).
- This project has no automated test framework (confirmed in `2026-08-15-tip-a-vet-venmo-rework-design.md`: "Automated tests (none exist; none added)"). Verification steps in this plan use direct SQL queries, `curl`, the Stripe CLI, and manual browser click-throughs in place of a test runner — treat each as the "run it and confirm" step for that task.
- The existing mock veterans in `data.js` (Jimmy Doyle, Eleanor Marsh, etc.) are pitch-mockup fiction and must **not** be seeded into the real `veterans` table — once real money is involved, only real, verified veterans may appear. The directory starts empty until the first admin approval.
- `view-served.js` / `SERVED_VETERANS` ("Vets We've Tipped") is explicitly out of scope for this plan — it is not touched.

```sql
create table public.veterans (

  branch text not null,
  era text not null,
  bio text not null,
  avatar_initials text not null,

  reviewed_at timestamptz,
  reviewed_by text,
  constraint at_least_one_handle check (

  )

create table public.tips (
  id uuid primary key default gen_random_uuid(),
  veteran_id uuid not null references public.veterans(id),
  amount numeric not null check (amount > 0),
  note text,
  donation_amount numeric not null default 0,
  stripe_checkout_session_id text,
  donation_status text not null default 'none' check (donation_status in ('none', 'pending', 'paid')),
  self_reported_sent boolean not null default false,
  created_at timestamptz not null default now()
);

create table public.admins (
  email text primary key
);

create view public.veteran_totals as
  select veteran_id, sum(amount) as total_tipped
  from public.tips
  where self_reported_sent = true
  group by veteran_id;

alter table public.veterans enable row level security;
alter table public.tips enable row level security;
alter table public.admins enable row level security;

grant usage on schema public to anon, authenticated;

grant select, insert on public.veterans to anon, authenticated;
grant update (status, reviewed_at, reviewed_by) on public.veterans to authenticated;

grant select, insert on public.tips to anon, authenticated;
```

```sql
create or replace function public.is_admin()
returns boolean
language sql
security definer
set search_path = public
as $$
  select exists (
    select 1 from public.admins where email = auth.jwt() ->> 'email'
  );
$$;

create policy "public can read approved veterans"
on public.veterans for select
to anon, authenticated
using (status = 'approved');

create policy "admins can read all veterans"
on public.veterans for select
to authenticated
using (public.is_admin());

create policy "anyone can submit a pending veteran"
on public.veterans for insert
to anon, authenticated
with check (status = 'pending');

create policy "admins can update veterans"
on public.veterans for update
to authenticated
using (public.is_admin())
with check (public.is_admin());

create policy "anyone can read tips"
on public.tips for select
to anon, authenticated
using (true);

create policy "anyone can create a tip"
on public.tips for insert
to anon, authenticated
with check (amount > 0);
```

```js
// supabase-client.js
const supabaseClient = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

async function fetchApprovedVeterans() {
  const { data: veterans, error } = await supabaseClient
    .from("veterans")
    .select("*")
    .eq("status", "approved")
    .order("submitted_at", { ascending: true });

  if (error) {
    throw error;
  }

  const { data: totals, error: totalsError } = await supabaseClient
    .from("veteran_totals")
    .select("veteran_id, total_tipped");

  if (totalsError) {
    throw totalsError;
  }

  const totalsByVetId = {};
  totals.forEach(function (row) {
    totalsByVetId[row.veteran_id] = row.total_tipped;
  });

  return veterans.map(function (vet) {
    return Object.assign({}, vet, {
      totalTipped: totalsByVetId[vet.id] || 0,
    });
  });
}

async function fetchVeteranById(id) {
  const { data: vet, error } = await supabaseClient
    .from("veterans")
    .select("*")
    .eq("status", "approved")
    .eq("id", id)
    .maybeSingle();

  if (error) {
    throw error;
  }
  if (!vet) {
    return null;
  }

  const { data: totalRow } = await supabaseClient
    .from("veteran_totals")
    .select("total_tipped")
    .eq("veteran_id", id)
    .maybeSingle();

  return Object.assign({}, vet, {
    totalTipped: (totalRow && totalRow.total_tipped) || 0,
  });
}
```

```js
async function submitVeteranForReview(fields) {
  let verificationDocPath = null;

  if (fields.verificationFile) {
    const filePath = crypto.randomUUID() + "-" + fields.verificationFile.name;
    const { error: uploadError } = await supabaseClient.storage
      .from("verification-docs")
      .upload(filePath, fields.verificationFile);

    if (uploadError) {
      throw uploadError;

  const { error } = await supabaseClient.from("veterans").insert({

    cashapp_handle: fields.cashappHandle || null,
    verification_doc_path: verificationDocPath,
```

```ts
// supabase/functions/create-donation-session/index.ts
import Stripe from "npm:stripe@14";
import { createClient } from "npm:@supabase/supabase-js@2";

const stripe = new Stripe(Deno.env.get("STRIPE_SECRET_KEY")!, {
  apiVersion: "2023-10-16",
});

const supabaseAdmin = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const { veteranId, tipAmount, donationAmount, note, returnUrl } = await req.json();

    if (!veteranId || !tipAmount || !(donationAmount > 0) || !returnUrl) {
      return new Response(
        JSON.stringify({ error: "veteranId, tipAmount, donationAmount, and returnUrl are required" }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } }
      );
    }

    const { data: tipRow, error: insertError } = await supabaseAdmin
      .from("tips")
      .insert({
        veteran_id: veteranId,
        amount: tipAmount,
        note: note || null,
        donation_amount: donationAmount,
        donation_status: "pending",
        self_reported_sent: false,
      })
      .select()
      .single();

    if (insertError) {
      throw insertError;
    }

    const session = await stripe.checkout.sessions.create({
      mode: "payment",
      client_reference_id: tipRow.id,
      line_items: [
        {
          price_data: {
            currency: "usd",
            product_data: { name: "Support Tip a Vet" },
            unit_amount: Math.round(donationAmount * 100),
          },
          quantity: 1,
        },
      ],
      success_url: returnUrl + "?tip_id=" + tipRow.id,
      cancel_url: returnUrl + "?tip_id=" + tipRow.id,
    });

    await supabaseAdmin
      .from("tips")
      .update({ stripe_checkout_session_id: session.id })
      .eq("id", tipRow.id);

    return new Response(JSON.stringify({ url: session.url, tipId: tipRow.id }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  } catch (error) {
    return new Response(JSON.stringify({ error: error.message }), {
      status: 500,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }
});
```

## Task 10: Stripe webhook Edge Function

**Files:**

**Interfaces:**
- Consumes: `public.tips` table, `client_reference_id` set by Task 9.
- Produces: HTTP endpoint `POST /functions/v1/stripe-webhook`, called by Stripe (not by the frontend). Flips `tips.donation_status` to `paid`.

- [ ] **Step 1: Write the function**

```ts
// supabase/functions/stripe-webhook/index.ts
import Stripe from "npm:stripe@14";
import { createClient } from "npm:@supabase/supabase-js@2";

const stripe = new Stripe(Deno.env.get("STRIPE_SECRET_KEY")!, {
  apiVersion: "2023-10-16",
});

const supabaseAdmin = createClient(
  Deno.env.get("SUPABASE_URL")!,
  Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
);

const webhookSecret = Deno.env.get("STRIPE_WEBHOOK_SECRET")!;

Deno.serve(async (req) => {
  const signature = req.headers.get("stripe-signature");
  const body = await req.text();

  let event;
  try {
    event = await stripe.webhooks.constructEventAsync(body, signature!, webhookSecret);
  } catch (err) {
    return new Response("Webhook signature verification failed: " + err.message, { status: 400 });
  }

  if (event.type === "checkout.session.completed") {
    const session = event.data.object as Stripe.Checkout.Session;
    const tipId = session.client_reference_id;

    if (tipId) {
      await supabaseAdmin
        .from("tips")
        .update({ donation_status: "paid" })
        .eq("id", tipId);
    }
  }

  return new Response(JSON.stringify({ received: true }), {
    headers: { "Content-Type": "application/json" },
  });
});
```

```js
async function createPendingTip(fields) {
  const { data, error } = await supabaseClient
    .from("tips")
    .insert({
      veteran_id: fields.veteranId,
      amount: fields.amount,
      note: fields.note || null,
      donation_amount: 0,
      donation_status: "none",
      self_reported_sent: false,
    })
    .select()
    .single();

  if (error) {
    throw error;
  }
  return data;
}

async function markTipSent(tipId) {
  const { error } = await supabaseClient
    .from("tips")
    .update({ self_reported_sent: true })
    .eq("id", tipId);

  if (error) {
    throw error;
  }
}

async function fetchTipById(tipId) {
  const { data, error } = await supabaseClient
    .from("tips")
    .select("*")
    .eq("id", tipId)
    .single();

  if (error) {
    throw error;
  }
  return data;
}

async function startDonationCheckout(fields) {
  const { data, error } = await supabaseClient.functions.invoke("create-donation-session", {
    body: {
      veteranId: fields.veteranId,
      tipAmount: fields.amount,
      donationAmount: fields.donationAmount,
      note: fields.note,
      returnUrl: fields.returnUrl,
    },
  });

  if (error) {
    throw error;
  }
  return data;
}
```

```js
function cashappWebLink(handle, amount) {

function paypalWebLink(handle, amount) {

function renderPaymentComplete(vetId, tipId) {
  const content = document.getElementById("payment-content");
  content.innerHTML = "<p>Loading…</p>";

  Promise.all([fetchVeteranById(vetId), fetchTipById(tipId)]).then(function (results) {
    const vet = results[0];
    const tip = results[1];
    if (!vet || !tip) {
      content.innerHTML = "<p>Something went wrong loading your tip.</p>";
      return;
    }
    showHandoffScreen(vet, tip.id, tip.amount, tip.note);
```
